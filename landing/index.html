Get Started in 3 Lines

Simple Python API. Production-ready in minutes. 

from btut import Simulator

# Simulate 100,000 agents
sim = Simulator(agents=100000, gamma=1.5)
result = sim.run()

print(f"Cooperation: {result.final_cooperation:.2%}")
# Output: Cooperation: 60.00%
